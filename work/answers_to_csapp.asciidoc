int saturating_add(int x, int y)
{
    unsigned r = (unsigned)x + y;
    int w = sizeof(int) << 3;
    bool x_is_minus = (x & INT_MIN);
    bool y_is_minus = (y & INT_MIN);

    int both_minus = (x_is_minus && y_is_minus);
    int both_pos = (!x_is_minus && !y_is_minus);

    // ~0 if neg_overflow, 0 otherwise
    unsigned neg_overflow = ~((both_minus && !(r & INT_MIN)) - 1);

    // ~0 if pos_overflow, 0 otherwise
    int pos_overflow = ~((both_pos && (r & INT_MIN)) - 1);
    // ~0 is !(neg_overflow || pos_overflow), 0 otherwise
    int no_overflow = ~(neg_overflow | pos_overflow);

    int TMax = (1 << (w - 1)) - 1;
    int TMin = -(1 << (w - 1));

    return (pos_overflow & TMax) | (neg_overflow & TMin) | (no_overflow & r);
}
